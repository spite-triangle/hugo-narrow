"\n" .Inner -}} {{- $lines := split (trim $inner "\n") "\n" -}} {{/* 逐行解析（状态机），见 helpers/term-line.html */}} {{- $s := newScratch -}} {{- $s.Set "blocks" slice -}} {{- $s.Set "curRows" slice -}} {{- $s.Set "copyParts" slice -}} {{- $s.Set "inCmd" false -}} {{- $s.Set "prevSlash" false -}} {{- range $lines -}} {{- $line := . -}} {{- $isCmd := hasPrefix $line "$" -}} {{- $isCont := and ($s.Get "inCmd") ($s.Get "prevSlash") -}} {{- if $isCmd -}} {{/* 新命令开始，先闭合上一条命令 */}} {{- if $s.Get "inCmd" -}} {{- $s.Set "blocks" ($s.Get "blocks" | append (dict "type" "cmd-group" "copy" (delimit ($s.Get "copyParts") " ") "rows" ($s.Get "curRows"))) -}} {{- end -}} {{- $parsed := partial "helpers/term-line.html" (dict "text" $line "prompt" true) -}} {{- $s.Set "curRows" (slice (dict "prompt" true "text" $parsed.text "comment" $parsed.comment)) -}} {{- $s.Set "copyParts" (slice $parsed.copy) -}} {{- $s.Set "inCmd" true -}} {{- $s.Set "prevSlash" $parsed.slash -}} {{- else if $isCont -}} {{/* 续行，归入当前命令 */}} {{- $parsed := partial "helpers/term-line.html" (dict "text" $line "prompt" false) -}} {{- $s.Set "curRows" ($s.Get "curRows" | append (dict "prompt" false "text" $parsed.text "comment" $parsed.comment)) -}} {{- $s.Set "copyParts" ($s.Get "copyParts" | append $parsed.copy) -}} {{- $s.Set "prevSlash" $parsed.slash -}} {{- else -}} {{/* 输出行，先闭合当前命令 */}} {{- if $s.Get "inCmd" -}} {{- $s.Set "blocks" ($s.Get "blocks" | append (dict "type" "cmd-group" "copy" (delimit ($s.Get "copyParts") " ") "rows" ($s.Get "curRows"))) -}} {{- $s.Set "inCmd" false -}} {{- end -}} {{- $s.Set "blocks" ($s.Get "blocks" | append (dict "type" "out" "text" $line)) -}} {{- $s.Set "prevSlash" false -}} {{- end -}} {{- end -}} {{/* 闭合末尾未结束的命令 */}} {{- if $s.Get "inCmd" -}} {{- $s.Set "blocks" ($s.Get "blocks" | append (dict "type" "cmd-group" "copy" (delimit ($s.Get "copyParts") " ") "rows" ($s.Get "curRows"))) -}} {{- end -}} {{- $blocks := $s.Get "blocks" -}} {{- .Page.Store.Set "hasTerm" true -}}
